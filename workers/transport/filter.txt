# PBF prefilter for `transport.yml` and `paths.yml` (`osmium tags-filter --expressions`).
# Restrictions come along: `tags-filter` keeps whole objects. Railways are the `railways` package.
# Wider than the schema on purpose; `workers/lint/transport.py` guards the other way.

# roads, from motorways to steps
w/highway

# ferries (on the Danube the only way across)
w/route=ferry

# aerialways and lifts
w/aerialway

# addresses: points and buildings with a house number
nwr/addr:housenumber
nwr/addr:conscriptionnumber

# built-up areas, which a road's speed context is measured against (`speed.py`)
wr/landuse=residential,commercial,retail

# traffic signs; `signs.py` names their images in place
n/traffic_sign
n/traffic_sign:forward
n/traffic_sign:backward
n/highway=stop,give_way,traffic_signals,speed_camera
