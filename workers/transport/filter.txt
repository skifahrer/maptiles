# PBF prefilter for `transport.yml` (`osmium tags-filter --expressions`): road OR ferry OR aerialway.
# Restrictions come along: `tags-filter` keeps whole objects. Railways are the `railways` package.
# Wider than the schema on purpose; `workers/lint/transport.py` guards the other way.

# roads, from motorways to steps
w/highway

# ferries (on the Danube the only way across)
w/route=ferry

# aerialways and lifts
w/aerialway

# addresses: points and buildings with a house number
nwr/addr:housenumber
nwr/addr:conscriptionnumber
