# Predfilter PBF pre `water.yml` (`osmium tags-filter --expressions`).
#
# Jeden priechod: vodstvo je „tok ALEBO vodná plocha ALEBO pobrežie".
#
# `r/` je tu nutné: veľké jazerá, priehrady a rieky so šírkou sú multipolygóny,
# ktorých členovia vlastné `natural=water` nemajú. Filter len na `w/` by pustil
# relácie bez geometrie a po Domaši by ticho neostalo nič.
#
# Zoznam je zámerne širší než schéma. Opačne to neplatí – stráži to
# `workers/lint/water.py`.

# --- toky: rieky, potoky, kanály, jarky ---
w/waterway
r/waterway

# --- vodné plochy: jazerá, priehrady, rybníky, zálivy, prielivy ---
n/natural=water
w/natural=water
r/natural=water
w/natural=bay
r/natural=bay
w/natural=strait
r/natural=strait
w/natural=spring
n/natural=spring

# --- more: pobrežná čiara ---
# Plocha oceánu z OSM nevzniká (skladá sa z pobrežných čiar celej planéty);
# tú kreslí základná mapa z `water_polygons` Planetileru. Tu je čiara, po
# ktorej more končí, a mená morí a zálivov.
w/natural=coastline

# --- vodné nádrže mapované ako využitie plochy ---
w/landuse=reservoir
r/landuse=reservoir
w/landuse=basin
r/landuse=basin
