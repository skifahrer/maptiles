# PBF prefilter for `water.yml` (`osmium tags-filter --expressions`): stream OR water OR coast.
# `r/` is needed: big lakes and reservoirs are multipolygons of untagged ways.
# Wider than the schema on purpose; `workers/lint/water.py` guards the other way.

# streams: rivers, brooks, canals, ditches
w/waterway
r/waterway

# water areas: lakes, reservoirs, ponds, bays, straits
n/natural=water
w/natural=water
r/natural=water
w/natural=bay
r/natural=bay
w/natural=strait
r/natural=strait
w/natural=spring
n/natural=spring

# sea: the coastline line; the base map draws the ocean area
w/natural=coastline

# reservoirs mapped as land use
w/landuse=reservoir
r/landuse=reservoir
w/landuse=basin
r/landuse=basin
