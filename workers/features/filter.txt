# PBF prefilter for `features.yml` and `points.yml` (`osmium tags-filter`).
# Wider than the schemas on purpose; `workers/lint/features.py` guards the other way.
# `nwr/` = nodes, ways and relations (osmium pulls in multipolygon members).

# embankments and cuttings
nwr/man_made=embankment,cutline,pipeline,spoil_heap,tower,mast,water_tower,chimney,adit,mineshaft,survey_point
nwr/embankment
nwr/cutting

# barriers as lines (point gates are in the OpenMapTiles POI layer)
nwr/barrier=wall,city_wall,retaining_wall,fence,guard_rail,handrail,hedge

# power lines
nwr/power=line,minor_line

# terrain edges and vegetation
nwr/natural=tree_row,gully,earth_bank,shingle,spring,cave_entrance,sinkhole

# water structures and waterfalls
nwr/waterway=dam,weir,lock_gate,waterfall

# areas the schema lacks
nwr/amenity=parking
nwr/landuse=landfill,farmyard,greenhouse_horticulture,brownfield

# monuments and mining heritage beyond the OpenMapTiles `historic` three
nwr/historic=archaeological_site,battlefield,fort,manor,tomb,wayside_cross,wayside_shrine,memorial,boundary_stone,milestone,charcoal_pile,mine,mine_shaft,mineshaft,adit

# huts and mountain passes
nwr/tourism=wilderness_hut
nwr/mountain_pass

# mountain rescue points and proposed roads (OpenMapTiles has no `proposed` class)
nwr/highway=emergency_access_point,proposed

# aerialways out of service (OpenMapTiles has those in service)
w/aerialway=construction,proposed,disused,abandoned
w/construction:aerialway
w/proposed:aerialway
w/disused:aerialway
w/abandoned:aerialway
w/razed:aerialway

# ski pistes, cross-country, ski touring
nwr/piste:type
