# Predfilter PBF pre vyhľadávací index (`osmium tags-filter --expressions`).
#
# Nadmnožina features filtra: všetko, čo kreslí mapu, je aj hľadateľné, a k tomu
# osídlenia a menované trasy – všetko, čo má titul.
#
# Zoznam je zámerne širší než schéma; presné rozhodovanie robí Python skript.
#
# `nwr/` = uzly, cesty aj relácie (osmium k nim dotiahne členov).

# --- násypy a zárezy (všetko, čo features kresli) ---
nwr/man_made=embankment,cutline,pipeline,spoil_heap,tower,mast,water_tower,chimney,adit,mineshaft,survey_point
nwr/embankment
nwr/cutting

# --- prekážky ako línie ---
nwr/barrier=wall,city_wall,retaining_wall,fence,guard_rail,handrail,hedge

# --- elektrické vedenie ---
nwr/power=line,minor_line

# --- terénne hrany a vegetácia ---
nwr/natural=tree_row,gully,earth_bank,shingle,spring,cave_entrance,sinkhole

# --- vodné stavby a vodopády ---
nwr/waterway=dam,weir,lock_gate,waterfall

# --- plochy, ktoré schéma nemá ---
nwr/amenity=parking
nwr/landuse=landfill,farmyard,greenhouse_horticulture,brownfield

# --- pamiatky mimo `historic` trojice ---
nwr/historic=archaeological_site,battlefield,fort,manor,tomb,wayside_cross,wayside_shrine,memorial,boundary_stone,milestone,charcoal_pile,mine,mine_shaft,mineshaft,adit

# --- útulne a horské priechody ---
nwr/tourism=wilderness_hut
nwr/mountain_pass

# --- núdzové body horskej služby ---
nwr/highway=emergency_access_point,proposed

# --- zjazdovky, bežky, skialp ---
nwr/piste:type

# --- OSÍDLENIA (nadmnožina features, z OpenMapTiles base layer) ---
nwr/place=city,town,village,hamlet,neighbourhood,suburb
nwr/place=locality,isolated_dwelling

# --- VRCH-Y A BODY ZÁUJMU (nadmnožina, z OpenMapTiles POI layer) ---
# Vrcholky a prirodzené bodové prvky hľadania – pôžička z OMT
nwr/natural=peak,saddle,cave,sinkhole,spring
nwr/tourism=alpine_hut,mountain_hut,wilderness_hut,camp_site,picnic_site,viewpoint
nwr/amenity=bar,cafe,restaurant,hotel,guest_house,hostel,memorial,ruins,fountain
nwr/man_made=cross,wayside_cross,wayside_shrine

# --- POMENOVANÉ TRASY (multi-way relations) ---
# Turistické chodníky, cyklotrasy, bežky, jazdecké trasy – jedna línia
# na trasu, titul z `name` alebo `name:sk`. Relácie typu route.
r/type=route
r/route=hiking,foot,bicycle,mtb,ski,horse,via_ferrata,walking

# --- POMENOVANÉ CESTY (pre jednotlivé línie mimo trás) ---
# Cesty s nadanou hodnotou `name` alebo `name:sk` – trvalý chodník,
# cyklotrasa bez relácie, alebo historická značka
nwr/highway=path,footway,track,steps,cycleway
nwr/name
